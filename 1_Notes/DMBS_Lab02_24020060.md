**Câu 4:**
```
CREATE OR REPLACE VIEW actor_list AS
SELECT
	a.actor_id,
	a.first_name,
	a.last_name,
	a.last_update,
	f.film_id,
	f.title
FROM actor a
JOIN film_actor fa ON a.actor_id = fa.actor_id
JOIN film AS f ON fa.film_id = f.film_id;

SELECT *
FROM actor_list;
```

**Câu 5:**
```
CREATE OR REPLACE VIEW sales_by_customer AS
SELECT c.customer_id, COUNT(p.amount) doanh_so
FROM customer c
JOIN payment p USING (customer_id)
GROUP BY c.customer_id
ORDER BY doanh_so DESC;


SELECT *
FROM sales_by_customer;
```
